My Original Array Code

function myExample() {
    let myTemp = Number(
        document.getElementById('myText01').value
    );

    let mySum = "";

    let myArray = new Array();
    myArray[0] = 'Fred';
    myArray[1] = 'Mary';
    myArray[2] = 'Lizzy';

    for (let myLoop = 0; myLoop < myTemp; myLoop++) {
        mySum += 'Loop #: ' + myLoop
            + ', Array Entry: ' + myArray[myLoop] + '<br>';
    }

    document.getElementById('myDiv01').innerHTML = mySum;
}
