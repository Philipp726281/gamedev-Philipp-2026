3. How do they work?

Simple Sort

Finds the smallest number in the unsorted part and moves it to the next position.

Bubble Sort

Compares two neighboring numbers. If the first is bigger, they switch places. This repeats until everything is sorted.

Quick Sort

Chooses a pivot number. Smaller numbers go to the left and bigger numbers go to the right. Then it repeats this process on both sides.

4. Simple Code Examples

Simple Sort

let numbers = [5, 2, 8, 1, 4];

for (let i = 0; i < numbers.length; i++) {
    for (let j = i + 1; j < numbers.length; j++) {

        if (numbers[i] > numbers[j]) {
            let temp = numbers[i];
            numbers[i] = numbers[j];
            numbers[j] = temp;
        }

    }
}

console.log(numbers);

Bubble Sort

let numbers = [5, 2, 8, 1, 4];

for (let i = 0; i < numbers.length - 1; i++) {

    for (let j = 0; j < numbers.length - 1 - i; j++) {

        if (numbers[j] > numbers[j + 1]) {

            let temp = numbers[j];
            numbers[j] = numbers[j + 1];
            numbers[j + 1] = temp;

        }
    }
}

console.log(numbers);

Quick Sort

function quickSort(numbers) {

    if (numbers.length <= 1) {
        return numbers;
    }

    let pivot = numbers[0];

    let smaller = numbers.slice(1).filter(n => n <= pivot);
    let bigger = numbers.slice(1).filter(n => n > pivot);

    return [
        ...quickSort(smaller),
        pivot,
        ...quickSort(bigger)
    ];
}

console.log(quickSort([5, 2, 8, 1, 4]));
